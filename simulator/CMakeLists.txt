cmake_minimum_required(VERSION 3.24)
project(ukraine_alerts_simulator LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(FetchContent)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/../include/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(lvgl URL https://github.com/lvgl/lvgl/archive/refs/tags/v9.4.0.tar.gz)
FetchContent_MakeAvailable(lvgl)
set(ENABLE_CJSON_TEST OFF CACHE BOOL "" FORCE)
FetchContent_Declare(cjson URL https://github.com/DaveGamble/cJSON/archive/refs/tags/v1.7.19.tar.gz)
FetchContent_MakeAvailable(cjson)
add_library(alerts_core
    ../src/model.cpp ../src/primary.cpp ../src/reserve.cpp
    ../src/engine.cpp ../src/demo.cpp ../src/time.cpp
    ../src/ui.cpp ../src/ui_map.cpp ../src/ui_details.cpp
    ../generated/map_assets.cpp
    ../generated/ui_font_18.c ../generated/ui_font_22.c ../generated/ui_font_30.c)
target_include_directories(alerts_core PUBLIC ../include ${cjson_SOURCE_DIR})
target_link_libraries(alerts_core PUBLIC cjson lvgl::lvgl)
target_compile_options(alerts_core PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_executable(model_tests ../tests/model_tests.cpp)
target_link_libraries(model_tests PRIVATE alerts_core)
add_test(NAME model_contract COMMAND model_tests)
add_executable(ui_tests ../tests/ui_tests.cpp)
target_link_libraries(ui_tests PRIVATE alerts_core)
add_test(NAME ui_contract COMMAND ui_tests)
add_executable(engine_tests ../tests/engine_tests.cpp)
target_link_libraries(engine_tests PRIVATE alerts_core)
add_test(NAME engine_contract COMMAND engine_tests)
add_test(NAME configuration_contract COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_LIST_DIR}/../tests/config_tests.cmake")
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
target_link_libraries(ui_tests PRIVATE PkgConfig::SDL2)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
add_executable(alerts_simulator main.cpp window.cpp connection.cpp)
target_link_libraries(alerts_simulator PRIVATE alerts_core PkgConfig::SDL2 CURL::libcurl Threads::Threads)
target_compile_options(alerts_simulator PRIVATE -Wall -Wextra -Werror)
include(../cmake/BuildConfig.cmake)
target_include_directories(alerts_simulator PRIVATE "${CMAKE_BINARY_DIR}/config")
