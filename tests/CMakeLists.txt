cmake_minimum_required(VERSION 3.24)
project(alerts_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(ENABLE_CJSON_TEST OFF CACHE BOOL "" FORCE)
FetchContent_Declare(cjson URL https://github.com/DaveGamble/cJSON/archive/refs/tags/v1.7.19.tar.gz)
FetchContent_MakeAvailable(cjson)
enable_testing()
add_executable(model_tests model_tests.cpp ../src/model.cpp ../src/primary.cpp ../src/reserve.cpp ../src/time.cpp)
target_include_directories(model_tests PRIVATE ../include ${cjson_SOURCE_DIR})
target_link_libraries(model_tests PRIVATE cjson)
target_compile_options(model_tests PRIVATE -Wall -Wextra -Werror -Wconversion)
add_test(NAME model_contract COMMAND model_tests)
